% TOC's \normalfont override below (\small here = \fontsize{10}{12}).

% chapter as both arguments shows the running head on both sides.

% both marks); also suppress the footer/page number on them.

% \emph does not toggle shape when already in italic/slanted context
\makeatletter

  \else
    {\itshape #1}%
  \fi}
\makeatother

% Force all TOC entries to use the sans-serif font via tocbasic hooks,
% which fire around the actual entry rendering (after \normalfont resets)
\makeatletter
\BeforeStartingTOC[toc]{%
  \pdfbookmark[0]{\contentsname}{toc}%
  \let\toc@@save@@normalfont\normalfont
  \renewcommand{\normalfont}{\toc@@save@@normalfont\sffamily\fontsize{12}{15}\selectfont}%
  \sffamily\fontsize{12}{15}\selectfont
  % Pages ending right before a \part render the footer in a wrong
  % (serif) font for unclear reasons; hide header/footer across the TOC.
  \thispagestyle{empty}%
  \pagestyle{empty}%
}
\AfterStartingTOC[toc]{%
  \let\normalfont\toc@@save@@normalfont
  \thispagestyle{empty}%
  \pagestyle{scrheadings}%
}
\makeatother

% The sans TOC font is wider than tocnumwidth was sized for, causing
% double-digit numbers (10. onward) to collide with the entry title.
\RedeclareSectionCommand[tocnumwidth=2.3em]{chapter}
\RedeclareSectionCommand[tocnumwidth=3.2em]{section}
\RedeclareSectionCommand[tocnumwidth=4.1em]{subsection}
